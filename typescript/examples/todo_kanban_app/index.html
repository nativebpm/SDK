const workflow = new Workflow('kanban-task-lifecycle', 'Kanban Task Lifecycle');

workflow
  .user('todo', 'Task in Backlog', ut => {
    ut.candidateGroups('developers');
  })
  .user('inProgress', 'Work on Task', ut => {
    ut.assignee('${developer}');
  })
  .user('review', 'Code Review', ut => {
    ut.candidateGroups('reviewers');
  })
  .when(v('approved').eq(false))
  .then(flow => {
    flow.service('notifyRejection', 'Notify Rejection', 'alerts')
        .user('inProgress', 'Work on Task'); // Back-edge
  })
  .else(flow => {
    flow.service('notifyApproval', 'Notify Approval', 'alerts')
        .end('done', 'Task Completed');
  });

  When(nativebpm.V("approved").Eq(false)).
  Then(func(flow *nativebpm.Branch) {
    flow.Service("notifyRejection", "Notify Rejection", "alerts").
         User("inProgress", "Work on Task", nativebpm.M{})
  }).
  Else(func(flow *nativebpm.Branch) {
    flow.Service("notifyApproval", "Notify Approval", "alerts").
         End("done", "Task Completed")
  })
const flow = new Workflow('itil-incident-management', 'Incident Lifecycle');

flow
  .user('triage', 'L1 Triage & Claim', ut => {
    ut.candidateGroups('l1_agents');
    ut.boundaryTimer('reactionSLA', timer => {
      timer.escalate('l2_escalation', () => {
        flow.user('l2_support', 'L2 Support', ut2 => {
          ut2.candidateGroups('l2_engineers');
        });
      });
    });
  })
  .user('investigate', 'Investigate & Resolve', ut => {
    ut.assignee('${agent}');
    ut.boundaryTimer('resolutionSLA', timer => {
      timer.service('alertManager', 'escalate');
    });
  })
  .user('review', 'Customer Verification')
  .when(v('reopen').eq(true))
  .then(branch => {
    branch.service('reopenTicket')
          .user('l2_support', 'L2 Support'); // Back-edge to L2
  })
  .else(branch => {
    branch.end('closed', 'Incident Closed');
  });
